SHOW DATABASES;

USE vitc;

select * from worker;

select worker_id, first_name, department from worker where salary <= 100000;

select  first_name, department, salary from worker where worker_id in (1,3,5,7,9);

select  first_name, department, salary from worker where worker_id  not in (0,2,4,6);

select * from worker where salary between 100000 and 400000;

select * from worker where salary not between 100000 and 400000;

select * from worker where salary between 300000 and 800000 and worker_id in (1,4,7) ;

select * from worker where first_name like '_i%';

select * from worker where first_name like '_a%';

select * from worker where last_name like '%a';

select * from worker where first_name not like 'a%';

select max(salary) from worker ;

select avg(salary) from worker where salary >= 100000;

select min(salary) from worker where department = 'admin' and first_name not like '%a%';

select first_name from worker
union 
select last_name from worker;

select first_name from worker
union all
select first_name from worker;

select distinct(joining_date) from worker order by joining_date;

select worker_id, first_name ,
CASE
when last_name in ('singh', 'sharma', 'kumar', 'bhati','chauhan') then "North Indian"
else 'South India'
end as 'Region' from worker;

create view Firsthr as select min(joining_date) from worker where department = "hr";

select * from firsthr;
